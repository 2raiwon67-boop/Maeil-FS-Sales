// doGet - GET 요청 처리 (담당자 목록 조회 / 네이버 장소 검색)
// ============================================================
function doGet(e) {
  const action = e.parameter.action;

  // --- 1. 담당자 목록 조회 ---
  if (action === 'getManagers') {
    try {
      const ss = SpreadsheetApp.getActiveSpreadsheet();
      const sheet = ss.getSheetByName('담당자관리');
      if (!sheet) {
        return jsonResponse({ success: false, error: '담당자관리 시트 없음' });
      }

      const data = sheet.getDataRange().getValues();
      const managers = [];
      for (let i = 1; i < data.length; i++) {
        const region = data[i][0];
        const name   = data[i][1];
        const email  = data[i][2];
        if (name) managers.push({ region: region || '', name, email: email || '' });
      }
      return jsonResponse({ success: true, managers });
    } catch (err) {
      return jsonResponse({ success: false, error: err.toString() });
    }
  }

  // --- 2. 네이버 장소 검색 프록시 ---
  if (action === 'naverSearch') {
    const query = e.parameter.query;
    if (!query) return jsonResponse({ success: false, error: '검색어 없음' });

    try {
      const url = 'https://openapi.naver.com/v1/search/local.json?query='
                  + encodeURIComponent(query) + '&display=5&sort=comment';

      const response = UrlFetchApp.fetch(url, {
        method: 'GET',
        headers: {
          'X-Naver-Client-Id': NAVER_CLIENT_ID,
          'X-Naver-Client-Secret': NAVER_CLIENT_SECRET
        },
        muteHttpExceptions: true
      });

      const code = response.getResponseCode();
      if (code !== 200) {
        return jsonResponse({ success: false, error: 'Naver API 오류: ' + code });
      }

      const result = JSON.parse(response.getContentText());
      const items = (result.items || []).map(item => ({
        title:       item.title.replace(/<[^>]+>/g, ''),
        category:    item.category,
        address:     item.address,
        roadAddress: item.roadAddress,
        telephone:   item.telephone,
        mapx:        item.mapx,
        mapy:        item.mapy
      }));

      return jsonResponse({ success: true, items });
    } catch (err) {
      return jsonResponse({ success: false, error: err.toString() });
    }
  }

  return jsonResponse({ success: false, error: '알 수 없는 action: ' + action });
}

// JSON 응답 헬퍼
function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

// doPost
// ============================================================
function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName('인허가데이터');

    // --- 1. 상태 변경 (updateStatus) ---
    if (data.action === 'updateStatus') {
      const sheetData = sheet.getDataRange().getValues();
      for (var i = 1; i < sheetData.length; i++) {
        if (sheetData[i][2] === data.name && sheetData[i][6] === data.address) {
          sheet.getRange(i + 1, 4).setValue(data.status);
          return jsonResponse({ success: true, message: 'Status updated' });
        }
      }
      return jsonResponse({ success: false, error: 'Target not found' });
    }

    // --- 2. 최종본 업로드 (upload) ---
    if (data.action === 'upload') {
      if (!data.data || !Array.isArray(data.data)) {
        return jsonResponse({ success: false, error: '데이터 형식 오류' });
      }

      const lastRow = sheet.getLastRow();
      const startNo = lastRow;
      const rows = data.data.map((row, index) => [
        startNo + index,
        row['영업 허가일']        || '',
        row['사업장명']           || '',
        row['거래여부(기입예정)']  || '',
        row['업태구분명']         || '',
        row['평형']              || '',
        row['도로명전체주소']     || '',
        row['주소1']             || '',
        row['주소2']             || '',
        row['주소3']             || '',
        row['순위']              || '',
        row['담당자']            || '',
        row['앱시트등록일']       || '',
        row['위도']              || '',
        row['경도']              || ''
      ]);

      if (rows.length > 0) {
        sheet.getRange(lastRow + 1, 1, rows.length, 15).setValues(rows);
      }

      const managerConfig  = loadManagerConfig(ss);
      const allRows        = sheet.getDataRange().getValues();
      const allDataObjects = [];

      for (let i = 1; i < allRows.length; i++) {
        const r = allRows[i];
        allDataObjects.push({
          '영업 허가일':        r[1],
          '사업장명':           r[2],
          '거래여부(기입예정)':  r[3],
          '업태구분명':         r[4],
          '평형':              r[5],
          '도로명전체주소':     r[6],
          '주소1':             r[7],
          '주소2':             r[8],
          '주소3':             r[9],
          '순위':              String(r[10]),  // 숫자→문자열 변환
          '담당자':            r[11],
          '앱시트등록일':       r[12],
          '위도':              r[13],
          '경도':              r[14]
        });
      }

      const result = sendEmailsByManager(allDataObjects, managerConfig);
      return jsonResponse({
        success:    true,
        count:      rows.length,
        emailsSent: result.count > 0,
        sentCount:  result.count,
        logs:       result.logs
      });
    }

    // --- 3. 담당자 이메일 매칭 (matchManagerEmail) ---
    if (data.action === 'matchManagerEmail') {
      const name  = data.name;
      const email = data.email;

      if (!name || !email) {
        return jsonResponse({ success: false, error: '이름 또는 이메일 없음' });
      }

      const managerSheet = ss.getSheetByName('담당자관리');
      if (!managerSheet) {
        return jsonResponse({ success: false, error: '담당자관리 시트 없음' });
      }

      const sheetData = managerSheet.getDataRange().getValues();
      for (let i = 1; i < sheetData.length; i++) {
        if (sheetData[i][1] === name) {
          managerSheet.getRange(i + 1, 3).setValue(email);
          Logger.log('담당자 이메일 자동 매칭: ' + name + ' → ' + email);
          return jsonResponse({ success: true, matched: true, name, email });
        }
      }

      return jsonResponse({ success: true, matched: false, name });
    }

    // 알 수 없는 action 처리
    return jsonResponse({ success: false, error: '알 수 없는 action: ' + data.action });

  } catch (error) {
    return jsonResponse({ success: false, error: error.toString() });
  }
}

// 담당자 설정 로드 ('담당자관리' 시트)
// ============================================================
function loadManagerConfig(ss) {
  const sheet = ss.getSheetByName('담당자관리');
  if (!sheet) {
    throw new Error("'담당자관리' 시트를 찾을 수 없습니다.");
  }

  const data            = sheet.getDataRange().getValues();
  const managerEmailMap = {};
  const branchManagers  = [];

  for (let i = 1; i < data.length; i++) {
    const region = data[i][0];
    const name   = data[i][1];
    const email  = data[i][2];

    if (!name || !email) continue;

    if (region === '지점장' || region === '전체') {
      branchManagers.push(email);
    } else {
      managerEmailMap[name] = email;
    }
  }

  return { managerEmailMap, branchManagers };
}

// 담당자별 분리 이메일 발송
// ============================================================
function sendEmailsByManager(allData, config) {
  const allTargets = filterEmailTargets(allData);
  if (allTargets.newObj.length === 0 && allTargets.revisitObj.length === 0) {
    Logger.log('이메일 발송 대상 없음 (D+14, D+28 조건 만족 건수 0)');
    return { count: 0, logs: allTargets.logs };
  }

  let visitLogs = [];
  try {
    visitLogs = loadVisitLogs();
    Logger.log('방문일지 로드 완료: ' + visitLogs.length + '건 (AI 분석에 활용)');
  } catch (e) {
    Logger.log('방문일지 로드 실패 (AI 분석 없이 진행): ' + e.toString());
  }

  const { managerEmailMap, branchManagers } = config;
  let totalSent = 0;

  const managersInNew     = new Set(allTargets.newObj.map(t => t.manager));
  const managersInRevisit = new Set(allTargets.revisitObj.map(t => t.manager));
  const activeManagers    = new Set([...managersInNew, ...managersInRevisit]);

  activeManagers.forEach(managerName => {
    if (!managerName || managerName === '미지정') return;

    const email = managerEmailMap[managerName];
    if (!email) {
      Logger.log('이메일 정보 없음: ' + managerName);
      return;
    }

    const myNew     = allTargets.newObj.filter(t => t.manager === managerName);
    const myRevisit = allTargets.revisitObj.filter(t => t.manager === managerName);

    if (myNew.length > 0 || myRevisit.length > 0) {
      const myTargets = { newObj: myNew, revisitObj: myRevisit };
      let aiStrategy  = null;
      try {
        aiStrategy = generateAIStrategyForManager(managerName, myTargets, visitLogs);
        if (aiStrategy) Logger.log(managerName + ' AI 전략 생성 완료');
      } catch (e) {
        Logger.log(managerName + ' AI 전략 생성 실패: ' + e.toString());
      }
      sendSummaryEmail(email, myTargets, managerName, aiStrategy);
      totalSent++;
    }
  });

  branchManagers.forEach(email => {
    let branchAiStrategy = null;
    try {
      branchAiStrategy = generateAIStrategyForManager('전체', allTargets, visitLogs);
    } catch (e) {
      Logger.log('지점장 AI 전략 생성 실패: ' + e.toString());
    }
    sendSummaryEmail(email, allTargets, '전체', branchAiStrategy);
    totalSent++;
  });

  return { count: totalSent, logs: allTargets.logs };
}

// 이메일 발송 (단일 수신자)
// ============================================================
function sendSummaryEmail(recipientEmail, targets, managerName, aiStrategy) {
  const newCount     = targets.newObj.length;
  const revisitCount = targets.revisitObj.length;

  const subject = EMAIL_CONFIG.subjectTemplate
    .replace('{newCount}', newCount)
    .replace('{revisitCount}', revisitCount);

  const htmlBody = generateSummaryEmailHTML(targets, aiStrategy);

  try {
    GmailApp.sendEmail(recipientEmail, subject, 'HTML 지원 클라이언트에서 확인해주세요.', {
      htmlBody: htmlBody,
      name:     EMAIL_CONFIG.senderName,
      from:     EMAIL_CONFIG.fromEmail   // ← 발신 주소 적용
    });
  } catch (e) {
    Logger.log('이메일 발송 실패: ' + recipientEmail + ' / ' + e.toString());
  }
}

// D+14 / D+28 대상 필터링
// ============================================================
function filterEmailTargets(data) {
  const now   = new Date();
  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());

  const newTargets     = [];
  const revisitTargets = [];
  let debugLogs        = [];

  data.forEach(row => {
    if (row['순위'] !== '1' && row['순위'] !== '2') return;

    let permitDateStr = row['영업 허가일'];
    if (!permitDateStr) return;

    // ★ Sheets 날짜 셀은 Date 객체로 반환됨 → 문자열로 변환
    if (permitDateStr instanceof Date) {
      const y = permitDateStr.getFullYear();
      const m = String(permitDateStr.getMonth() + 1).padStart(2, '0');
      const d = String(permitDateStr.getDate()).padStart(2, '0');
      permitDateStr = y + '-' + m + '-' + d;
    }

    // '.'이 포함된 문자열이면 '-'로 변환 (2024.02.05 → 2024-02-05)
    if (typeof permitDateStr === 'string' && permitDateStr.includes('.')) {
      permitDateStr = permitDateStr.replace(/\./g, '-');
    }

    try {
      const parts = permitDateStr.toString().split(/[-.]/);
      let permitDate;

      if (parts.length === 3) {
        const y = parseInt(parts[0], 10);
        const m = parseInt(parts[1], 10) - 1;
        const d = parseInt(parts[2], 10);
        permitDate = new Date(y, m, d);
      } else {
        permitDate = new Date(permitDateStr);
      }

      if (isNaN(permitDate.getTime())) {
        Logger.log('날짜 파싱 실패: ' + row['영업 허가일']);
        return;
      }

      permitDate.setHours(0, 0, 0, 0);

      const timeDiff = today.getTime() - permitDate.getTime();
      const daysDiff = Math.floor(timeDiff / (1000 * 60 * 60 * 24));

      Logger.log('[' + row['사업장명'] + '] ' + permitDateStr + ' -> D+' + daysDiff + '일 경과');

      const item = {
        name:       row['사업장명'],
        pyeong:     row['평형'],
        permitDate: permitDateStr,
        address:    row['도로명전체주소'],
        status:     row['거래여부(기입예정)'],
        manager:    row['담당자'],
        lat:        row['위도'],
        lon:        row['경도'],
        nearby:     findNearbyStores(row, data)
      };

      if (daysDiff >= 14 || daysDiff >= 28) {
        debugLogs.push('[' + row['사업장명'] + '] 날짜:' + permitDateStr + ' (D+' + daysDiff + '), 상태:' + row['거래여부(기입예정)']);
      }

      // D+14 이상: '인허가' 상태만
      if (daysDiff >= 14 && row['거래여부(기입예정)'] === '인허가') {
        newTargets.push(item);
      }
      // D+28 이상: '공사중' 상태만
      else if (daysDiff >= 28 && row['거래여부(기입예정)'] === '공사중') {
        revisitTargets.push(item);
      }
    } catch (e) {
      debugLogs.push('[ERROR] ' + row['사업장명'] + ' : ' + e.toString());
    }
  });

  if (debugLogs.length > 20) {
    debugLogs = debugLogs.slice(0, 20);
    debugLogs.push('... (too many logs)');
  }

  return { newObj: newTargets, revisitObj: revisitTargets, logs: debugLogs };
}

// 방문일지 시트에서 방문 데이터 로드
function loadVisitLogs() {
  try {
    const ss = SpreadsheetApp.openById(VISIT_LOGS_SHEET_ID);
    let sheet = ss.getSheets()[0];
    for (const s of ss.getSheets()) {
      if (s.getSheetId() === VISIT_LOGS_SHEET_GID) { sheet = s; break; }
    }
    const data = sheet.getDataRange().getValues();
    if (data.length < 2) return [];
    const headers = data[0];
    return data.slice(1).map(row => {
      const obj = {};
      headers.forEach((h, i) => { obj[h] = row[i] || ''; });
      return obj;
    });
  } catch (e) {
    Logger.log('방문일지 로드 실패: ' + e.toString());
    return [];
  }
}

// 특정 거래처의 방문 이력 조회 (최근 5건)
function getVisitHistoryForStore(storeName, visitLogs) {
  if (!storeName || !visitLogs || visitLogs.length === 0) return [];
  const name = storeName.trim();
  return visitLogs.filter(log => {
    const place = (log['방문처(거래처)'] || '').toString().trim();
    return place.includes(name) || name.includes(place.substring(0, 4));
  }).slice(-5);
}

// Gemini API 호출
function callGeminiAPI(prompt) {
  try {
    const payload = {
      contents:         [{ parts: [{ text: prompt }] }],
      generationConfig: { temperature: 0.7, maxOutputTokens: 600, topP: 0.95 }
    };
    const response = UrlFetchApp.fetch(GEMINI_API_URL + '?key=' + GEMINI_API_KEY, {
      method:             'POST',
      contentType:        'application/json',
      payload:            JSON.stringify(payload),
      muteHttpExceptions: true
    });
    if (response.getResponseCode() !== 200) {
      Logger.log('Gemini API 오류: ' + response.getResponseCode() + ' / ' + response.getContentText().substring(0, 200));
      return null;
    }
    const result = JSON.parse(response.getContentText());
    return (result.candidates && result.candidates[0] && result.candidates[0].content &&
            result.candidates[0].content.parts && result.candidates[0].content.parts[0])
           ? result.candidates[0].content.parts[0].text : null;
  } catch (e) {
    Logger.log('Gemini API 호출 실패: ' + e.toString());
    return null;
  }
}

// 담당자별 AI 방문 전략 생성
function generateAIStrategyForManager(managerName, targets, visitLogs) {
  const allTargets = [...targets.newObj, ...targets.revisitObj];
  if (allTargets.length === 0 || visitLogs.length === 0) return null;

  const targetData = allTargets.map(t => {
    const nearbyVisits = [];
    if (t.nearby && t.nearby.length > 0) {
      t.nearby.forEach(nearby => {
        const history = getVisitHistoryForStore(nearby.name, visitLogs);
        if (history.length > 0) {
          nearbyVisits.push({
            store:        nearby.name,
            dist:         nearby.dist + 'km',
            recentVisits: history.map(v => ({
              date:    (v['작성일'] || v['일정기간'] || '').toString().substring(0, 10),
              manager: v['작성자'] || '',
              summary: (v['내용'] || '').toString().substring(0, 80),
              status:  v['거래상태'] || ''
            }))
          });
        }
      });
    }
    return { name: t.name, status: t.status, nearbyWithHistory: nearbyVisits };
  });

  const hasHistory = targetData.some(t => t.nearbyWithHistory.length > 0);
  if (!hasHistory) return null;

  const prompt = '당신은 경기북부 FS 영업팀의 AI 전략 비서입니다.\n' +
    '담당자 ' + managerName + '의 오늘 방문 대상 인근 거래처 방문 이력을 분석하여 실용적인 현장 전략을 제안해주세요.\n\n' +
    '# 방문 대상 및 인근 거래처 방문 이력\n' +
    JSON.stringify(targetData, null, 2) + '\n\n' +
    '# 요청사항\n' +
    '각 방문 대상별로 인근 거래처 방문 이력을 참고하여:\n' +
    '1. 지역 내 영업 분위기 (이전 방문 내용 기반)\n' +
    '2. 함께 방문하면 효율적인 인근 거래처와 그 이유\n' +
    '3. 이전 방문자의 경험에서 얻을 수 있는 영업 팁\n\n' +
    '# 답변 규칙\n' +
    '- 각 방문처마다 2-3줄로 핵심만 간결하게\n' +
    '- 인근 이력이 있는 경우에만 해당 내용 포함\n' +
    '- 전체 400자 이내\n' +
    '- 이모지 활용하여 구분 명확히\n\n' +
    '답변:';

  return callGeminiAPI(prompt);
}

// ============================================================
// 인근 매장 검색
// ============================================================
function calculateDistance(lat1, lon1, lat2, lon2) {
  if (!lat1 || !lon1 || !lat2 || !lon2) return 9999;
  const R    = 6371;
  const dLat = (lat2 - lat1) * Math.PI / 180;
  const dLon = (lon2 - lon1) * Math.PI / 180;
  const a    = Math.sin(dLat/2) * Math.sin(dLat/2) +
               Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
               Math.sin(dLon/2) * Math.sin(dLon/2);
  return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
}

function findNearbyStores(target, allData) {
  const nearby    = [];
  const targetLat = parseFloat(target['위도']);
  const targetLon = parseFloat(target['경도']);

  allData.forEach(row => {
    if (row['사업장명'] === target['사업장명']) return;
    if (String(row['순위']) !== '1') return;
    const dist = calculateDistance(targetLat, targetLon, parseFloat(row['위도']), parseFloat(row['경도']));
    if (dist <= 3.0) {
      nearby.push({ name: row['사업장명'], dist: dist.toFixed(1), category: row['업태구분명'] });
    }
  });

  return nearby.sort((a, b) => a.dist - b.dist).slice(0, 3);
}
